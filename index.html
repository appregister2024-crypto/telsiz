Şu anda çalışan bu arayüz, tarayıcılar arası BroadcastChannel ve Web Audio API ile aynı makinedeki birden çok sekmede gerçek iki telsiz gibi anında çalışır. Telefon ve bilgisayarlar arasında çalıştırmak için aşağıdaki Node.js sunucu kodunu çalıştırabilirsiniz.

1 Sunucu Bilgisayarın Yerel IP Adresini Bulma

Telsiz sunucusunu çalıştıracağınız ana bilgisayarda terminali açın:

# Windows için:
ipconfig
# IPv4 Address kısmını bulun (Örn: 192.168.1.50)

# macOS / Linux için:
ifconfig | grep "inet " | grep -v 127.0.0.1
# veya: ip a

2 Telefonda Mikrofon Erişimi İçin HTTPS Zorunluluğu

Mobil Chrome ve Safari, güvenli olmayan HTTP üzerinden mikrofon izni vermez (yalnızca `localhost` hariç). Yerel IP'de (`192.168.x.x`) mikrofonu açmak için mkcert ile 1 dakikada yerel SSL sertifikası üretilir:

npm install -g mkcert
mkcert create-ca
mkcert create-cert --domains "localhost,192.168.1.50"
# Oluşan cert.crt ve cert.key dosyalarını server klasörüne koyun.

3 Üretim Hazır Node.js + Socket.IO Sunucu Kodu (`server.js`)

Aynı anda yalnızca tek bir kullanıcının konuşmasını garantileyen sunucu kilit mekanizması ve sinyalizasyon:

const express = require('express');
const https = require('https');
const fs = require('fs');
const { Server } = require('socket.io');

const app = express();
app.use(express.static('public'));

// HTTPS Sertifikaları (Yerel Ağda mobil mikrofon erişimi için)
const server = https.createServer({
  key: fs.readFileSync('./cert.key'),
  cert: fs.readFileSync('./cert.crt')
}, app);

const io = new Server(server, { cors: { origin: '*' } });

// Odalar ve aktif konuşma kilitleri
const rooms = new Map(); // roomId -> { currentSpeaker: socketId, timer: NodeJS.Timeout, members: Map }

io.on('connection', (socket) => {
  let userRoom = null;
  let userName = 'Bilinmeyen';

  socket.on('join_room', ({ roomId, username }) => {
    userRoom = roomId;
    userName = username;
    socket.join(roomId);

    if (!rooms.has(roomId)) {
      rooms.set(roomId, { currentSpeaker: null, speakerTimeout: null, members: new Map() });
    }
    const room = rooms.get(roomId);
    room.members.set(socket.id, username);

    // Bildirim gönder
    io.to(roomId).emit('room_roster', {
      members: Array.from(room.members.entries()).map(([id, name]) => ({ id, name })),
      currentSpeaker: room.currentSpeaker
    });
  });

  // Bas-Konuş Kilidi Talep Etme
  socket.on('request_floor', () => {
    if (!userRoom || !rooms.has(userRoom)) return;
    const room = rooms.get(userRoom);

    // Kilit boş mu kontrol et
    if (room.currentSpeaker === null) {
      room.currentSpeaker = socket.id;
      
      // 15 saniye güvenlik kilidi zaman aşımı (takılı kalan mikrofon önlemi)
      room.speakerTimeout = setTimeout(() => {
        releaseFloor(userRoom, socket.id);
      }, 15000);

      io.to(userRoom).emit('floor_granted', { speakerId: socket.id, speakerName: userName });
    } else {
      socket.emit('floor_denied', { reason: 'Kanal meşgul' });
    }
  });

  // Buton bırakıldığında kilidi çöz
  socket.on('release_floor', () => {
    if (!userRoom) return;
    releaseFloor(userRoom, socket.id);
  });

  // WebRTC Sinyalizasyon İletimi (P2P Mesh)
  socket.on('signal', ({ targetId, data }) => {
    io.to(targetId).emit('signal', { senderId: socket.id, data });
  });

  socket.on('disconnect', () => {
    if (userRoom && rooms.has(userRoom)) {
      const room = rooms.get(userRoom);
      room.members.delete(socket.id);
      if (room.currentSpeaker === socket.id) {
        releaseFloor(userRoom, socket.id);
      }
      io.to(userRoom).emit('room_roster', {
        members: Array.from(room.members.entries()).map(([id, name]) => ({ id, name })),
        currentSpeaker: room.currentSpeaker
      });
      if (room.members.size === 0) rooms.delete(userRoom);
    }
  });

  function releaseFloor(roomId, requesterId) {
    const room = rooms.get(roomId);
    if (!room || room.currentSpeaker !== requesterId) return;
    clearTimeout(room.speakerTimeout);
    room.currentSpeaker = null;
    io.to(roomId).emit('floor_released');
  }
});

const PORT = 3000;
server.listen(PORT, '0.0.0.0', () => {
  console.log(`📡 Telsiz Sunucusu Hazır: https://0.0.0.0:${PORT}`);
});

4 Güvenlik Duvarı & Sorun Giderme

  • Port İzni: Sunucu makinenin güvenlik duvarında 3000 portuna (TCP) ve WebRTC için UDP trafiğine izin verin.
  • AP İzolasyonu (İstemci İzolasyonu): Bazı otel veya misafir Wi-Fi yönlendiricilerinde "Client Isolation" açıktır. Cihazlar birbirini göremezse ev/ofis normal Wi-Fi ağına bağlanın.
